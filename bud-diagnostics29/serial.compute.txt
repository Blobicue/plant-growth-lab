#version 450
layout(local_size_x=64) in;
layout(set=0,binding=0,std430) readonly buffer A { uvec2 a[]; };
layout(set=0,binding=1,std430) buffer B { uvec2 b[]; };
layout(set=0,binding=2,std430) readonly buffer R { uint req[]; };
layout(set=0,binding=3,std430) buffer O { uint outp[]; };
layout(push_constant,std430) uniform P { uint mode; uint count; } p;
const uint X=21,Y=8,Z=13,COUNT=X*Y*Z,CAP=100000000;
ivec2 flux(ivec2 d){return (d/100)*ivec2(3,2)+(d-(d/100)*100)*ivec2(3,2)/100;}
void main(){
 uint i=gl_GlobalInvocationID.x;
 if(p.mode==0){
  if(i>=COUNT)return;
  uint x=i%X,z=(i/X)%Z,y=i/(X*Z);ivec2 v=ivec2(a[i]),change=ivec2(0);
  if(x>0)change+=flux(ivec2(a[i-1])-v);
  if(x+1<X)change+=flux(ivec2(a[i+1])-v);
  if(z>0)change+=flux(ivec2(a[i-X])-v);
  if(z+1<Z)change+=flux(ivec2(a[i+X])-v);
  if(y>0)change+=flux(ivec2(a[i-X*Z])-v);
  if(y+1<Y)change+=flux(ivec2(a[i+X*Z])-v);
  b[i]=uvec2(v+change);return;
 }
 if(i!=0)return;
 uint count=req[0],queries=req[1];
 for(uint e=0;e<count;e++){
  uint o=4+e*4,index=req[o];uvec2 want=uvec2(req[o+1],req[o+2]);bool deposit=req[o+3]==1;
  uvec2 actual=min(want,deposit?uvec2(CAP)-b[index]:b[index]);
  if(deposit)b[index]+=actual;else b[index]-=actual;
  outp[8+e*2]=actual.x;outp[9+e*2]=actual.y;
 }
 for(uint q=0;q<queries;q++){
  uvec2 value=b[req[4+count*4+q]];outp[8+count*2+q*2]=value.x;outp[9+count*2+q*2]=value.y;
 }
 uvec2 total=uvec2(0),carry=uvec2(0),low=uvec2(CAP),high=uvec2(0);
 for(uint j=0;j<COUNT;j++){uvec2 old=total;total+=b[j];carry+=uvec2(lessThan(total,old));low=min(low,b[j]);high=max(high,b[j]);}
 outp[0]=total.x;outp[1]=total.y;outp[2]=low.x;outp[3]=low.y;outp[4]=high.x;outp[5]=high.y;outp[6]=carry.x;outp[7]=carry.y;
}
